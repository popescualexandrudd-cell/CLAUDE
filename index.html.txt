      function onRefreshFail() {
        document.getElementById('skeleton-view').style.display = 'none';
        document.getElementById('loader').style.display = 'none';

        if (!hasDashboard()) { backToLogin(); showLoginError('Eroare de conexiune la server. Încercați din nou.'); }

      function onRefresh(res) {
        if (!res || res.error) {
          if (res && res.error === 'SESIUNE_EXPIRATA') {
            Store.del('mastersToken'); Store.del('mastersCache');
            SESSION.token = '';
            backToLogin();
            showLoginError('Sesiunea a expirat. Autentificați-vă din nou.');
            return;
          }
          // Eroare temporară: dacă avem deja date pe ecran, le păstrăm — nu întrerupem părintele.
          document.getElementById('skeleton-view').style.display = 'none';
          document.getElementById('loader').style.display = 'none';
          document.getElementById('syncText').style.display = 'none';
          if (!hasDashboard()) { backToLogin(); showLoginError((res && res.error) || 'Conexiune indisponibilă.'); }
          return;
        }
        SESSION.gdprOk = !!res.gdprOk;
        SESSION.appUrl = res.appUrl || '';
        SESSION.installUrl = res.installUrl || '';
        if (res.anunt) aplicaAnunt(res.anunt);
        Store.set('mastersCache', JSON.stringify(res.data));
        renderDashboard(res.data, false);
        afterAuth();
      }

      /* ── Restaurarea sesiunii: token din URL (injectat de server) sau din memoria locală ── */
      function initSession() {
        if (BOOT && BOOT.token) {
          SESSION.token = BOOT.token;
          SESSION.gdprOk = !!BOOT.gdprOk;
          SESSION.appUrl = BOOT.appUrl || '';
          SESSION.installUrl = BOOT.installUrl || '';
          Store.set('mastersToken', BOOT.token);
          document.getElementById('login-view').style.display = 'none';
          if (BOOT.data && BOOT.data.length) {
            // Datele au venit odată cu pagina: zero apeluri, zero așteptare.
            Store.set('mastersCache', JSON.stringify(BOOT.data));
            renderDashboard(BOOT.data, false);
            afterAuth();
          } else {
            showSkeleton();
            google.script.run.withSuccessHandler(onRefresh).withFailureHandler(onRefreshFail).portalRefresh(BOOT.token);
          }
          return true;
        }

        var t = Store.get('mastersToken');
        if (t) {
          SESSION.token = t;
          document.getElementById('login-view').style.display = 'none';
          var cached = Store.get('mastersCache'), rendered = false;
          if (cached) {
            try { renderDashboard(JSON.parse(cached), true); rendered = true; } catch (e) {}
          }
          if (!rendered) showSkeleton();
          google.script.run.withSuccessHandler(onRefresh).withFailureHandler(onRefreshFail).portalRefresh(t);
          return true;
        }
        return false;
      }

      function gdprVerifica(btn) {
        if (!SESSION.token) return;
        var textInitial = btn.innerText;
        btn.innerText = 'Se verifică…'; btn.disabled = true;
        gdprMesaj('', '');

        google.script.run.withSuccessHandler(function (r) {
          btn.innerText = textInitial; btn.disabled = false;

          // Sesiune expirată: nu e o problemă de acord, ci de autentificare.
          if (r && r.error === 'SESIUNE_EXPIRATA') {
            Store.del('mastersToken'); Store.del('mastersCache');
            SESSION.token = ''; gdprInchide();
            document.getElementById('athletes-container').innerHTML = '';
            backToLogin(); showLoginError('Sesiunea a expirat. Autentifică-te din nou.');
            return;
          }
          if (r && r.gdprOk) { SESSION.gdprOk = true; gdprInchide(); return; }

          gdprEsec(gdprIncercari < 1
            ? 'Încă nu găsim răspunsul tău. Dacă tocmai l-ai trimis, mai încearcă o dată în câteva secunde.'
            : 'Tot nu îl găsim. Probabil numele sportivului din formular diferă de cel din evidența clubului.');
        }).withFailureHandler(function () {
          btn.innerText = textInitial; btn.disabled = false;
          // Și eșecul de conexiune trebuie să conteze: altfel, offline, părintele
          // ar rămâne blocat pentru totdeauna, fără nicio cale de ieșire.
          gdprEsec('Nu am putut verifica (conexiune indisponibilă). Mai încearcă.');
        }).portalGdprVerifica(SESSION.token);
      }

      /* După două încercări fără succes — indiferent de motiv — părintele

      function gdprEsec(mesaj) {

      function generateCalendarLink(program, nume) {
          var prog = String(program || '').toUpperCase(), days = [];
          if (prog.indexOf('L-V') > -1) { days = ['MO','TU','WE','TH','FR']; } else if (prog.indexOf('L-J') > -1) { days = ['MO','TU','WE','TH']; } else {
              if (prog.indexOf('LUNI') > -1 || /\bL\b/.test(prog) || prog.indexOf('L/') > -1) days.push('MO');
              if (prog.indexOf('MARTI') > -1 || /\bMA\b/.test(prog) || prog.indexOf('MA/') > -1) days.push('TU');
              if (prog.indexOf('MIERCURI') > -1 || /\bMI\b/.test(prog) || prog.indexOf('MIE') > -1 || prog.indexOf('MI/') > -1) days.push('WE');
              if (prog.indexOf('JOI') > -1 || /\bJ\b/.test(prog) || prog.indexOf('J/') > -1) days.push('TH');
              if (prog.indexOf('VINERI') > -1 || /\bV\b/.test(prog) || prog.indexOf('V/') > -1) days.push('FR');
              if (prog.indexOf('SAMBATA') > -1 || /\bS\b/.test(prog)) days.push('SA');
              if (prog.indexOf('DUMINICA') > -1 || /\bD\b/.test(prog)) days.push('SU');
          }
          if(days.length === 0) days = ['MO','TU','WE','TH','FR']; 
          var startHour = "180000", endHour = "200000";
          var timeMatch = prog.match(/(\d{1,2})(?::(\d{2}))?\s*[-–]\s*(\d{1,2})(?::(\d{2}))?/);
          if (timeMatch) {
              var h1 = (timeMatch[1].length === 1 ? '0' : '') + timeMatch[1], m1 = timeMatch[2] || '00', h2 = (timeMatch[3].length === 1 ? '0' : '') + timeMatch[3], m2 = timeMatch[4] || '00';
              startHour = h1 + m1 + "00"; endHour = h2 + m2 + "00";
          }
          var d = new Date(), yyyy = d.getFullYear(), mm = (d.getMonth() + 1 < 10 ? '0' : '') + (d.getMonth() + 1), dd = (d.getDate() < 10 ? '0' : '') + d.getDate();
          var dateStr = yyyy + mm + dd, title = encodeURIComponent("🎾 Tenis - " + nume), details = encodeURIComponent("Antrenament de tenis la Club Masters."), loc = encodeURIComponent("Club Tenis Masters, Strada Gheorghe Petrașcu 43-45, 031592 București, România"), dates = dateStr + "T" + startHour + "/" + dateStr + "T" + endHour, rrule = "RRULE:FREQ=WEEKLY;BYDAY=" + days.join(',');
          return "https://calendar.google.com/calendar/render?action=TEMPLATE&text=" + title + "&dates=" + dates + "&recur=" + rrule + "&details=" + details + "&location=" + loc;
      }

      /* ───────────────────────────────────────────────────────────────────────
         PLATĂ RAPIDĂ
         Nu există un standard prin care o pagină web să deschidă aplicația
         bancară de pe telefon cu transferul deja completat — ar fi nevoie de un
         furnizor de inițiere de plăți (PSD2), cu contract și costuri.
         Ce se poate face fără nimic în plus:
           • cod QR în format EPC/SEPA, scanabil din majoritatea aplicațiilor
             bancare („Plată prin QR"), care completează singur beneficiarul,
             IBAN-ul, suma și detaliile;
           • copierea IBAN-ului și a referinței cu o apăsare.
         Codul QR se generează LOCAL, în telefon — datele nu pleacă nicăieri.
         ─────────────────────────────────────────────────────────────────────── */

      function sumaDinTarif(t) {
        var m = String(t == null ? '' : t).replace(',', '.').match(/(\d+(?:\.\d{1,2})?)/);
        return m ? m[1] : '';
      }

      /* Structura EPC069-12, acceptată de scanerele de plăți din aplicațiile bancare. */
      function epcPayload(nume, iban, suma, referinta) {
        return ['BCD', '002', '1', 'SCT', '',
                String(nume).substring(0, 70),
                String(iban).replace(/\s+/g, ''),
                suma ? ('RON' + suma) : '',
                '', '',
                String(referinta).substring(0, 140)].join('\n');
      }

      function copiaza(text, btn) {
        var reusit = false;
        try {
          var ta = document.createElement('textarea');
          ta.value = text;
          ta.style.position = 'fixed'; ta.style.opacity = '0';
          document.body.appendChild(ta); ta.focus(); ta.select();
          reusit = document.execCommand('copy');
          document.body.removeChild(ta);
        } catch (e) { reusit = false; }

        if (!reusit && navigator.clipboard) {
          navigator.clipboard.writeText(text).then(function () { copiatFeedback(btn); }, function () {});
          return;
        }
        if (reusit) copiatFeedback(btn);
        else prompt('Copiază manual:', text);
      }

      }

      /* Desenează codul QR după randare. Fără biblioteca QR, secțiunea rămâne
         funcțională — doar fără imagine. */
      function deseneazaQR() {
        if (typeof QRCode === 'undefined') {
          var lipsa = document.querySelectorAll('.qr-box');
          for (var k = 0; k < lipsa.length; k++) lipsa[k].style.display = 'none';
          return;
        }
        var noduri = document.querySelectorAll('.qr-target');
        for (var i = 0; i < noduri.length; i++) {
          var n = noduri[i];
          if (n.getAttribute('data-gata') === '1') continue;
          try {
            new QRCode(n, { text: n.getAttribute('data-payload'), width: 150, height: 150,
                            colorDark: '#111111', colorLight: '#ffffff' });
            n.setAttribute('data-gata', '1');
          } catch (e) { n.parentNode.style.display = 'none'; }
        }
      }

      function login() {
        var phone   = document.getElementById('phoneInput').value;
        var prenume = document.getElementById('prenumeInput').value;
        document.getElementById('errorMsg').style.display = 'none';

        if (!phone)   { showLoginError('Introduceți numărul de telefon.'); return; }
        if (!prenume) { showLoginError('Introduceți prenumele sportivului.'); return; }

        var remember = document.getElementById('rememberMe').checked;
        if (remember) { Store.set('mastersSavedPhone', phone); Store.set('mastersSavedName', prenume); }
        else { Store.del('mastersSavedPhone'); Store.del('mastersSavedName'); }

        document.getElementById('login-view').style.display = 'none';
        showSkeleton();

        // UN SINGUR apel: autentificare + token + date complete.
        google.script.run.withSuccessHandler(function(response) {
            if (!response || response.error) {
                backToLogin();
                showLoginError((response && response.error) || 'Eroare necunoscută.');
                return;
            }
            SESSION.token      = response.token;
            SESSION.gdprOk     = !!response.gdprOk;
            SESSION.appUrl     = response.appUrl || '';
            SESSION.installUrl = response.installUrl || '';
            if (response.anunt) aplicaAnunt(response.anunt);
            if (remember) Store.set('mastersToken', response.token);
            Store.set('mastersCache', JSON.stringify(response.data));
            renderDashboard(response.data, false);
            afterAuth();
        }).withFailureHandler(showError).portalLogin(phone, prenume, (navigator.userAgent || ''));
      }
